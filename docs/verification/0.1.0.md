## Native plugin evidence

Disposable integration tests were added under
`examples/catalog/integration_test/` only in a detached checkout of the release
candidate. The disposable catalog manifest added the Flutter SDK
`integration_test` package and direct `shared_preferences: 2.5.5` development
dependency. The complete test inputs are preserved in the reproduction appendix
below. Two separate `flutter test` processes ran from `examples/catalog` against
the native macOS target:

```sh
flutter test integration_test/release_evidence_test.dart \
  -d macos --dart-define=EVIDENCE_STAGE=select-dark
flutter test integration_test/release_evidence_test.dart \
  -d macos --dart-define=EVIDENCE_STAGE=relaunch-system
```

The first process cleared the evidence keys, launched `CatalogApp` with the
real `SharedPreferencesAsync` macOS implementation, selected dark through
`AppearanceSelector`, and read `dark` from the exact key
`flutter_foundation_catalog.appearance.mode.v1`. It wrote `light` through a
second `PreferencesAppearanceStore` using namespace `release_evidence_other`,
then proved the catalog namespace still read dark.

The first process exited. The second process started a newly built native app,
read dark before constructing the controller, launched the catalog, and verified
dark selection and resolution. It selected system through the UI and read
`system` from the same persisted key. It then delivered dark and light platform
brightness values through Flutter's platform dispatcher; the running native app
resolved each brightness while the stored value remained `system`. Both native
processes passed.

The available macOS backend did not expose a supported way to induce a real
storage outage. No native unavailable-storage result is claimed. Unit tests
exercise store/controller/selector failure and retry behavior with controlled
backends, but those are not presented as native plugin evidence.

Create a detached checkout at the release-candidate SHA. In its catalog

Create `examples/catalog/integration_test/release_evidence_test.dart` with this
exact content (SHA-256
`89c316975e7d9559fa8e57600d7082396535e41c7f6125ccaa659b4ab4989f31`):

```dart
import 'package:birb_appearance/birb_appearance.dart';
import 'package:flutter/material.dart';
import 'package:flutter_foundation_catalog/main.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:integration_test/integration_test.dart';
import 'package:shared_preferences/shared_preferences.dart';

const _catalogKey = 'flutter_foundation_catalog.appearance.mode.v1';
const _otherKey = 'release_evidence_other.appearance.mode.v1';
const _stage = String.fromEnvironment('EVIDENCE_STAGE');

void main() {
  IntegrationTestWidgetsFlutterBinding.ensureInitialized();

  testWidgets('real SharedPreferences release evidence stage $_stage', (
    tester,
  ) async {
    final preferences = SharedPreferencesAsync();
    final catalogStore = PreferencesAppearanceStore(
      applicationNamespace: 'flutter_foundation_catalog',
      preferences: preferences,
    );

    if (_stage == 'select-dark') {
      await preferences.remove(_catalogKey);
      await preferences.remove(_otherKey);
      await _pumpCatalog(tester, catalogStore);
      await _openAppearance(tester);
      await tester.tap(find.text('Dark'));
      await tester.pumpAndSettle();
      expect(await preferences.getString(_catalogKey), 'dark');
      expect(_brightness(tester), Brightness.dark);

      final otherStore = PreferencesAppearanceStore(
        applicationNamespace: 'release_evidence_other',
        preferences: preferences,
      );
      await otherStore.write(AppearanceMode.light);
      expect(await preferences.getString(_otherKey), 'light');
      expect(await catalogStore.read(), (
        mode: AppearanceMode.dark,
        isPersisted: true,
      ));
      return;
    }

    if (_stage == 'relaunch-system') {
      expect(await preferences.getString(_catalogKey), 'dark');
      await _pumpCatalog(tester, catalogStore);
      await _openAppearance(tester);
      expect(_brightness(tester), Brightness.dark);
      expect(find.text('Selected: dark'), findsOneWidget);

      await tester.tap(find.text('System'));
      await tester.pumpAndSettle();
      expect(await preferences.getString(_catalogKey), 'system');

      tester.platformDispatcher.platformBrightnessTestValue = Brightness.dark;
      await tester.pumpAndSettle();
      expect(_brightness(tester), Brightness.dark);
      expect(await preferences.getString(_catalogKey), 'system');

      tester.platformDispatcher.platformBrightnessTestValue = Brightness.light;
      await tester.pumpAndSettle();
      expect(_brightness(tester), Brightness.light);
      expect(await preferences.getString(_catalogKey), 'system');
      return;
    }

    fail('Unknown EVIDENCE_STAGE=$_stage');
  });
}

Future<void> _pumpCatalog(WidgetTester tester, AppearanceStore store) async {
  tester.view.physicalSize = const Size(1400, 1000);
  tester.view.devicePixelRatio = 1;
  addTearDown(tester.view.reset);
  await tester.pumpWidget(CatalogApp(appearanceStore: store));
  await tester.pumpAndSettle();
}

Future<void> _openAppearance(WidgetTester tester) async {
  await tester.tap(
    find.byKey(CatalogKeys.destination(CatalogSection.appearance)),
  );
  await tester.pumpAndSettle();
}

Brightness _brightness(WidgetTester tester) =>
    Theme.of(tester.element(find.byKey(CatalogKeys.root))).brightness;
```

Create `examples/catalog/integration_test/forced_preview_evidence_test.dart`
with this exact content (SHA-256
`a23406ef27cd10833add60ec2f3e6ed9a9c79f07b148f89094c130669dfe1774`):

```dart
import 'package:birb_appearance/birb_appearance.dart';
import 'package:birb_design_system/birb_design_system.dart';
import 'package:birb_design_system/design_system_preview.dart';
import 'package:flutter/material.dart';
import 'package:flutter/services.dart';
import 'package:flutter_foundation_catalog/main.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:integration_test/integration_test.dart';
import 'package:shared_preferences/shared_preferences.dart';

const _previewName = String.fromEnvironment('BIRB_THEME_PREVIEW');
const _catalogKey = 'flutter_foundation_catalog.appearance.mode.v1';

void main() {
  IntegrationTestWidgetsFlutterBinding.ensureInitialized();

  testWidgets('forced $_previewName native catalog inspection', (tester) async {
    final preview = CatalogThemePreview.fromEnvironment(_previewName);
    expect(preview, isNot(CatalogThemePreview.persisted));
    final preferences = SharedPreferencesAsync();
    final stored = preview == CatalogThemePreview.light ? 'dark' : 'light';
    await preferences.setString(_catalogKey, stored);
    final store = PreferencesAppearanceStore(
      applicationNamespace: 'flutter_foundation_catalog',
      preferences: preferences,
    );

    tester.view.physicalSize = const Size(1400, 1000);
    tester.view.devicePixelRatio = 1;
    addTearDown(tester.view.reset);
    await tester.pumpWidget(
      CatalogApp(appearanceStore: store, preview: preview),
    );
    await tester.pumpAndSettle();

    final materialApp = tester.widget<MaterialApp>(find.byType(MaterialApp));
    expect(materialApp.themeAnimationDuration, BirbDurations.instant);

    final expected = preview == CatalogThemePreview.light
        ? Brightness.light
        : Brightness.dark;
    expect(
      Theme.of(tester.element(find.byKey(CatalogKeys.root))).brightness,
      expected,
    );
    expect(await preferences.getString(_catalogKey), stored);

    FocusManager.instance.primaryFocus?.unfocus();
    await tester.sendKeyEvent(LogicalKeyboardKey.tab);
    await tester.pump();
    expect(FocusManager.instance.primaryFocus, isNotNull);

    await tester.tap(find.byKey(CatalogKeys.narrowToggle));
    await tester.tap(find.byKey(CatalogKeys.largeTextToggle));
    await tester.pump();
    expect(tester.getSize(find.byKey(CatalogKeys.contentViewport)).width, 320);
    final section = find.byKey(CatalogKeys.section(CatalogSection.components));
    expect(MediaQuery.textScalerOf(tester.element(section)).scale(10), 20);

    final dialogTrigger = find.byKey(
      BirbThemeHarnessKeys.fixture((
        family: BirbHarnessFamily.overlays,
        name: 'dialogTrigger',
      )),
    );
    await tester.ensureVisible(dialogTrigger);
    await tester.tap(dialogTrigger);
    await tester.pumpAndSettle();
    expect(find.byKey(BirbThemeHarnessKeys.dialog), findsOneWidget);
    await tester.tap(find.byKey(BirbThemeHarnessKeys.dialogDismiss));
    await tester.pumpAndSettle();

    await _select(tester, CatalogSection.codeReview);
    expect(find.byType(BirbReviewHarness), findsOneWidget);
    await _select(tester, CatalogSection.appearance);
    expect(find.byKey(CatalogKeys.forcedPreview), findsOneWidget);
    await _select(tester, CatalogSection.accessibility);
    await tester.drag(find.byType(ListView), const Offset(0, -600));
    await tester.pumpAndSettle();
    expect(find.byKey(CatalogKeys.errorSample), findsOneWidget);
    expect(find.byKey(CatalogKeys.disabledSample), findsOneWidget);
    expect(tester.takeException(), isNull);
    expect(await preferences.getString(_catalogKey), stored);
  });
}

Future<void> _select(WidgetTester tester, CatalogSection section) async {
  await tester.tap(find.byKey(CatalogKeys.destination(section)));
  await tester.pumpAndSettle();
}
```
